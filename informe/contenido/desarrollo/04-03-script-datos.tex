\section{Script SQL y datos de prueba realistas}

\subsection{Creaci\'on de esquemas y tablas}

El siguiente bloque DDL crea los cinco esquemas de negocio, las tablas normalizadas con todas sus restricciones declarativas y las relaciones foráneas con comportamientos \texttt{ON DELETE} apropiados.
El script completo y ejecutable se incluye en el Anexo A (\texttt{script\_bd2\_completo.sql}).

\begin{lstlisting}[language=SQL, caption={DDL: Esquemas, tablas de cat\'alogo y restricciones}, label=lst:ddl-catalogo]
-- =============================================
-- ESQUEMAS DE NEGOCIO
-- =============================================
CREATE SCHEMA IF NOT EXISTS catalogo;
CREATE SCHEMA IF NOT EXISTS ventas;
CREATE SCHEMA IF NOT EXISTS operaciones;
CREATE SCHEMA IF NOT EXISTS auditoria;

-- =============================================
-- ESQUEMA CATALOGO
-- =============================================
CREATE TABLE catalogo."Categorias" (
  "Id"     SERIAL PRIMARY KEY,
  "Nombre" VARCHAR(60)  NOT NULL,
  "Slug"   VARCHAR(80)  NOT NULL,
  CONSTRAINT uq_categorias_nombre UNIQUE ("Nombre"),
  CONSTRAINT uq_categorias_slug   UNIQUE ("Slug")
);

CREATE TABLE catalogo."Marcas" (
  "Id"     SERIAL PRIMARY KEY,
  "Nombre" VARCHAR(80) NOT NULL,
  "Pais"   VARCHAR(60),
  CONSTRAINT uq_marcas_nombre UNIQUE ("Nombre")
);

CREATE TABLE catalogo."Componentes" (
  "Id"          SERIAL PRIMARY KEY,
  "Nombre"      VARCHAR(200)   NOT NULL,
  "Descripcion" TEXT,
  "CategoriaId" INTEGER        NOT NULL,
  "MarcaId"     INTEGER        NOT NULL,
  "Precio"      NUMERIC(10,2)  NOT NULL,
  "Stock"       INTEGER        NOT NULL DEFAULT 0,
  "ImagenUrl"   VARCHAR(500),
  "Activo"      BOOLEAN        NOT NULL DEFAULT TRUE,
  "CreadoEn"    TIMESTAMPTZ    NOT NULL DEFAULT NOW(),
  CONSTRAINT ck_componentes_precio CHECK ("Precio" > 0),
  CONSTRAINT ck_componentes_stock  CHECK ("Stock"  >= 0),
  CONSTRAINT fk_comp_categoria
    FOREIGN KEY ("CategoriaId")
    REFERENCES catalogo."Categorias"("Id") ON DELETE RESTRICT,
  CONSTRAINT fk_comp_marca
    FOREIGN KEY ("MarcaId")
    REFERENCES catalogo."Marcas"("Id") ON DELETE RESTRICT
);

CREATE TABLE catalogo."Combos" (
  "Id"          SERIAL PRIMARY KEY,
  "Nombre"      VARCHAR(200)   NOT NULL,
  "Descripcion" TEXT,
  "PrecioBase"  NUMERIC(10,2)  NOT NULL,
  "Descuento"   NUMERIC(5,2)   NOT NULL DEFAULT 0,
  "Activo"      BOOLEAN        NOT NULL DEFAULT TRUE,
  CONSTRAINT ck_combos_precio     CHECK ("PrecioBase" > 0),
  CONSTRAINT ck_combos_descuento  CHECK ("Descuento" >= 0 AND "Descuento" <= 100)
);

CREATE TABLE catalogo."ComboComponentes" (
  "ComboId"      INTEGER NOT NULL,
  "ComponenteId" INTEGER NOT NULL,
  "Cantidad"     INTEGER NOT NULL,
  CONSTRAINT pk_combo_comp   PRIMARY KEY ("ComboId", "ComponenteId"),
  CONSTRAINT ck_cc_cantidad  CHECK ("Cantidad" > 0),
  CONSTRAINT fk_cc_combo
    FOREIGN KEY ("ComboId")
    REFERENCES catalogo."Combos"("Id") ON DELETE CASCADE,
  CONSTRAINT fk_cc_componente
    FOREIGN KEY ("ComponenteId")
    REFERENCES catalogo."Componentes"("Id") ON DELETE RESTRICT
);
\end{lstlisting}

\begin{lstlisting}[language=SQL, caption={DDL: Esquema ventas --- Pedidos y PedidoItems normalizados}, label=lst:ddl-ventas]
-- =============================================
-- ESQUEMA VENTAS
-- =============================================
CREATE TABLE ventas."Pedidos" (
  "Id"           SERIAL PRIMARY KEY,
  "UserId"       TEXT           NOT NULL,
  "FechaPedido"  TIMESTAMPTZ    NOT NULL DEFAULT NOW(),
  "Total"        NUMERIC(12,2)  NOT NULL,
  "MetodoPago"   SMALLINT       NOT NULL,
  "Estado"       SMALLINT       NOT NULL DEFAULT 0,
  "Notas"        TEXT,
  CONSTRAINT ck_pedidos_total   CHECK ("Total" >= 0),
  CONSTRAINT ck_pedidos_metodo  CHECK ("MetodoPago" IN (0,1,2,3)),
  CONSTRAINT ck_pedidos_estado  CHECK ("Estado" IN (0,1,2,3,4)),
  CONSTRAINT fk_pedidos_user
    FOREIGN KEY ("UserId")
    REFERENCES "AspNetUsers"("Id") ON DELETE RESTRICT
);

CREATE TABLE ventas."PedidoItems" (
  "Id"             SERIAL        PRIMARY KEY,
  "PedidoId"       INTEGER       NOT NULL,
  "ComponenteId"   INTEGER,
  "ComboId"        INTEGER,
  "Cantidad"       INTEGER       NOT NULL,
  "PrecioUnitario" NUMERIC(10,2) NOT NULL,
  CONSTRAINT ck_pi_cantidad  CHECK ("Cantidad" > 0),
  CONSTRAINT ck_pi_precio    CHECK ("PrecioUnitario" > 0),
  CONSTRAINT ck_pi_exclusivo CHECK (
    ("ComponenteId" IS NOT NULL AND "ComboId" IS NULL) OR
    ("ComboId"      IS NOT NULL AND "ComponenteId" IS NULL)
  ),
  CONSTRAINT fk_pi_pedido
    FOREIGN KEY ("PedidoId")
    REFERENCES ventas."Pedidos"("Id") ON DELETE CASCADE,
  CONSTRAINT fk_pi_componente
    FOREIGN KEY ("ComponenteId")
    REFERENCES catalogo."Componentes"("Id") ON DELETE RESTRICT,
  CONSTRAINT fk_pi_combo
    FOREIGN KEY ("ComboId")
    REFERENCES catalogo."Combos"("Id") ON DELETE RESTRICT
);

CREATE TABLE ventas."CarritosAbandonados" (
  "Id"           SERIAL      PRIMARY KEY,
  "UserId"       TEXT        NOT NULL,
  "ItemsJson"    JSONB       NOT NULL DEFAULT '[]'::jsonb,
  "FechaCreado"  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  "FechaExpira"  TIMESTAMPTZ,
  CONSTRAINT fk_carritos_user
    FOREIGN KEY ("UserId")
    REFERENCES "AspNetUsers"("Id") ON DELETE CASCADE
);
\end{lstlisting}
