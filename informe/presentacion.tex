% DIAPOSITIVA 4: NORMALIZACIÓN 1FN, 2FN Y 3FN
% ============================================================================
\begin{frame}{3. Proceso de Normalizaci\'on (1FN $\to$ 2FN $\to$ 3FN)}{Eliminaci\'on de Redundancias y Anomal\'ias}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.32\textwidth}
      \cardBox[updsNavy]{Primera Forma Normal (1FN)}{%
        \textbf{Anomal\'ia:} \texttt{ItemsJson} almacenado como cadena \texttt{TEXT} plana sin validaci\'on estructural.\\[4pt]
        \textbf{Soluci\'on:} Migraci\'on a \texttt{JSONB} binario validado con \'indice invertido \texttt{GIN} para consultas nativas \texttt{@>}.
      }
    \end{column}
    \begin{column}{0.32\textwidth}
      \cardBox[updsBlue]{Segunda Forma Normal (2FN)}{%
        \textbf{Anomal\'ia:} Atributos descriptivos repetidos en cat\'alogos compuestos.\\[4pt]
        \textbf{Soluci\'on:} Clave primaria compuesta \texttt{(ComboId, ComponenteId)} en \texttt{ComboComponentes} con dependencia total de \texttt{Cantidad}.
      }
    \end{column}
    \begin{column}{0.32\textwidth}
      \cardBox[updsTeal]{Tercera Forma Normal (3FN)}{%
        \textbf{Anomal\'ia:} \texttt{Tipo}/\texttt{Marca} en texto libre y \texttt{FullName} duplicado.\\[4pt]
        \textbf{Soluci\'on:} Extracci\'on de \texttt{Categorias} y \texttt{Marcas} con FK; eliminaci\'on de \texttt{FullName} y enlace a \texttt{Regiones}/\texttt{Sucursales}.
      }
    \end{column}
  \end{columns}

  \vspace{4mm}
  \cardBox[updsSlate]{Depuraci\'on Adicional del Modelo}{%
    Eliminaci\'on definitiva de \texttt{public."Products"} (scaffold legacy en ingl\'es) e incorporaci\'on de \texttt{ProveedorId}, \texttt{StockMinimo} y \texttt{EstadoObsolescencia} en \texttt{catalogo.Componentes}.
  }
\end{frame}
